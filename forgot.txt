    $result = $conn->query("SELECT * FROM users WHERE email='$email'");

        // Update password
        $update = $conn->query("UPDATE users SET password='$new_password' WHERE email='$email'");

            echo "<p style='color:green;'>Password updated successfully! You can now <a href='login.php'>log in</a>.</p>";
        } else {
            echo "<p style='color:red;'>Error updating password: " . $conn->error . "</p>";

        echo "<p style='color:red;'>Email not found in our records.</p>";
    }
}
